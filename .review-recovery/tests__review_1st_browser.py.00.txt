Default: render self-contained HTML in memory, deny external requests, and use
an explicitly synthetic localStorage object. This is not a navigation/hosting
or real broadcast verification. REVIEW_URL enables a normal local CI server.

def check(name,ok,detail=None):
    checks.append({'name':name,'passed':bool(ok),**({'detail':detail} if detail is not None else {})})
    assert ok,name+(': '+str(detail) if detail is not None else '')

def install_store(page,seed=None):
    page.evaluate('''(seed)=>{const store=new Map(Object.entries(seed||{}));
      Object.defineProperty(window,'localStorage',{configurable:true,value:{
        getItem:k=>store.has(k)?store.get(k):null,setItem:(k,v)=>store.set(k,String(v)),
        removeItem:k=>store.delete(k),clear:()=>store.clear()}});
      window.__testStore=store;}''',seed or {})

def ready(context,clock=False,seed=None):
    p=context.new_page();p.on('pageerror',lambda e:errors.append(str(e)))
    p.on('request',lambda r:requests.append(r.url) if r.url.startswith('http') else None)
    p.route(re.compile(r'^https?://'),lambda r:r.abort()) if not URL else p.route('https://**/*',lambda r:r.abort())
    if clock:p.clock.install(time=datetime(2026,11,4,13,0,tzinfo=timezone.utc))
    if URL:p.goto(URL,wait_until='load')
    else:
        install_store(p,seed)
        p.set_content(HTML,wait_until='load')
    p.evaluate('''()=>document.addEventListener('securitypolicyviolation',e=>{window.__violations??=[];window.__violations.push(e.violatedDirective)})''')
    expect(p.locator('#main h1')).to_be_visible()
    return p

def go(p,path):

    expect(p.locator('#main h1')).to_be_visible()
    # Hash navigation only; never navigates to another network address.
    p.wait_for_timeout(60)
